<script>
let provider, signer, userAddress;
const tokenAddress = "0x791055A7d52AA392eaE8De04250497f33807E46A";

const abi = [
  "function balanceOf(address owner) view returns (uint256)",
  "function decimals() view returns (uint8)"
];

let autoInterval = null;

async function connectWallet(){
  if(window.ethereum){
    provider = new ethers.providers.Web3Provider(window.ethereum);
    await provider.send("eth_requestAccounts", []);
    signer = provider.getSigner();
    userAddress = await signer.getAddress();

    document.getElementById("status").innerText = userAddress.slice(0,6)+"...";
  } else {
    alert("Install MetaMask");
  }
}

async function getBalance(){
  if(!provider) return alert("Connect wallet first");

  const contract = new ethers.Contract(tokenAddress, abi, provider);
  const balance = await contract.balanceOf(userAddress);
  const decimals = await contract.decimals();

  const formatted = ethers.utils.formatUnits(balance, decimals);
  document.getElementById("balance").innerText = formatted;
}

async function getPrice(){
  const res = await fetch("https://api.dexscreener.com/latest/dex/pairs/bsc/0xb90071e377a31a6ea2cfdebe19a4d5226c420b6b");
  const data = await res.json();

  const price = parseFloat(data.pair.priceUsd);
  document.getElementById("price").innerText = "$" + price.toFixed(4);
  return price;
}

function startAuto(){
  const target = parseFloat(document.getElementById("targetPrice").value);
  if(!target) return alert("Enter target price");

  document.getElementById("autoStatus").innerText = "Running...";

  autoInterval = setInterval(async ()=>{
    const current = await getPrice();
    if(current <= target){
      document.getElementById("autoStatus").innerText = "Target reached";
    }
  }, 5000);
}

function stopAuto(){
  clearInterval(autoInterval);
  document.getElementById("autoStatus").innerText = "Stopped";
}
</script>
